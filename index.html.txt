function formatSize(bytes) {
  if (bytes < 1024) return bytes + ' B';
  if (bytes < 1024 * 1024) return (bytes / 1024).toFixed(1) + ' KB';
  return (bytes / (1024 * 1024)).toFixed(1) + ' MB';
}

  if (!currentFile) return;

function sleep(ms) { return new Promise(r => setTimeout(r, ms)); }

function loadVideo(file) {
  return new Promise((resolve, reject) => {
    const v = document.createElement('video');
    v.src = URL.createObjectURL(file);
    v.muted = true;
    v.playsInline = true;
    v.preload = 'auto';

    const timeout = setTimeout(() => reject(new Error('Таймаут загрузки видео')), 30000);

    v.onloadedmetadata = () => {
      clearTimeout(timeout);
      if (!isFinite(v.duration) || v.duration < 0.1) {
        reject(new Error('Видео слишком короткое или повреждено'));
        return;
      }
      resolve(v);
    };
    v.onerror = () => {
      clearTimeout(timeout);
      reject(new Error('Браузер не смог открыть этот видеоформат. Попробуйте MP4 (H.264).'));
    };
  });
}

async function extractFrames(video, count, onProgress) {
  const canvas = document.createElement('canvas');
  canvas.width = FRAME_SIZE;
  canvas.height = FRAME_SIZE;
  const ctx = canvas.getContext('2d', { willReadFrequently: true });

  const dur = video.duration;
  const start = Math.min(0.1, dur * 0.02);
  const end = Math.max(start + 0.1, dur - 0.1);
  const times = [];
  for (let i = 0; i < count; i++) {
    times.push(start + (end - start) * (i / (count - 1)));
  }

  const frames = [];
  for (let i = 0; i < times.length; i++) {
    try {
      await seekTo(video, times[i]);
      ctx.drawImage(video, 0, 0, FRAME_SIZE, FRAME_SIZE);
      const img = ctx.getImageData(0, 0, FRAME_SIZE, FRAME_SIZE);
      frames.push(grayFromImageData(img));
    } catch (e) {}
    onProgress && onProgress(i, times.length);
  }
  return frames;
}

function seekTo(video, time) {
  return new Promise((resolve, reject) => {
    const cleanup = () => {
      video.removeEventListener('seeked', onSeeked);
      video.removeEventListener('error', onErr);
    };
    const onSeeked = () => { cleanup(); resolve(); };
    const onErr = () => { cleanup(); reject(new Error('seek error')); };
    video.addEventListener('seeked', onSeeked);
    video.addEventListener('error', onErr);
    if (Math.abs(video.currentTime - time) < 0.01) { cleanup(); resolve(); return; }
    try { video.currentTime = time; } catch (e) { cleanup(); reject(e); }
  });
}

function grayFromImageData(imgData) {
  const { data, width, height } = imgData;
  const out = new Float32Array(width * height);
  for (let i = 0, j = 0; i < data.length; i += 4, j++) {
    out[j] = (0.299 * data[i] + 0.587 * data[i+1] + 0.114 * data[i+2]) / 255;
  }
  return out;
}

function centerCrop(frame, frameSize, outSize) {
  const cx = frameSize / 2;
  const half = outSize / 2;
  const x0 = Math.max(0, Math.round(cx - half));
  const y0 = Math.max(0, Math.round(cx - half));
  const out = new Float32Array(outSize * outSize);
  for (let y = 0; y < outSize; y++) {
    for (let x = 0; x < outSize; x++) {
      out[y * outSize + x] = frame[(y0 + y) * frameSize + (x0 + x)];
    }
  }
  return out;
}

function fft1d(re, im) {
  const n = re.length;
  for (let i = 1, j = 0; i < n; i++) {
    let bit = n » 1;
    for (; j & bit; bit »= 1) j ^= bit;
    j ^= bit;
    if (i < j) {
      [re[i], re[j]] = [re[j], re[i]];
      [im[i], im[j]] = [im[j], im[i]];
    }
  }
  for (let len = 2; len <= n; len «= 1) {
    const half = len » 1;
    const ang = -2 * Math.PI / len;
    const wRe = Math.cos(ang), wIm = Math.sin(ang);
    for (let i = 0; i < n; i += len) {
      let curRe = 1, curIm = 0;
      for (let j = 0; j < half; j++) {


const uRe = re[i + j], uIm = im[i + j];
        const vRe = re[i + j + half] * curRe - im[i + j + half] * curIm;
        const vIm = re[i + j + half] * curIm + im[i + j + half] * curRe;
        re[i + j] = uRe + vRe;
        im[i + j] = uIm + vIm;
        re[i + j + half] = uRe - vRe;
        im[i + j + half] = uIm - vIm;
        const nRe = curRe * wRe - curIm * wIm;
        curIm = curRe * wIm + curIm * wRe;
        curRe = nRe;
      }
    }
  }
}

function fft2dMagnitude(gray, n) {
  const re = [], im = [];
  for (let y = 0; y < n; y++) {
    re.push(new Float64Array(n));
    im.push(new Float64Array(n));
    for (let x = 0; x < n; x++) re[y][x] = gray[y * n + x];
  }
  for (let y = 0; y < n; y++) fft1d(re[y], im[y]);
  const cr = new Float64Array(n), ci = new Float64Array(n);
  for (let x = 0; x < n; x++) {
    for (let y = 0; y < n; y++) { cr[y] = re[y][x]; ci[y] = im[y][x]; }
    fft1d(cr, ci);
    for (let y = 0; y < n; y++) { re[y][x] = cr[y]; im[y][x] = ci[y]; }
  }
  const half = n » 1;
  const mag = new Float32Array(n * n);
  for (let y = 0; y < n; y++) {
    for (let x = 0; x < n; x++) {
      const sy = (y + half) % n;
      const sx = (x + half) % n;
      mag[y * n + x] = Math.log(Math.hypot(re[sy][sx], im[sy][sx]) + 1e-6);
    }
  }
  return mag;
}

function fftFeatures(gray, n) {
  const mag = fft2dMagnitude(gray, n);
  const half = n » 1;
  const exclude = Math.max(6, n » 3);

  const vals = [];
  for (let y = 0; y < n; y++) {
    for (let x = 0; x < n; x++) {
      const dx = x - half, dy = y - half;
      if (dx * dx + dy * dy < exclude * exclude) continue;
      vals.push(mag[y * n + x]);
    }
  }
  if (vals.length === 0) return { peak: 0, std: 0 };

  let sum = 0, max = -Infinity;
  for (const v of vals) { sum += v; if (v > max) max = v; }
  const mean = sum / vals.length;

  let varSum = 0;
  for (const v of vals) varSum += (v - mean) * (v - mean);
  const std = Math.sqrt(varSum / vals.length);

  return { peak: max - mean, std };
}

function laplacianVar(gray, w, h) {
  let sum = 0, sum2 = 0, n = 0;
  for (let y = 1; y < h - 1; y++) {
    for (let x = 1; x < w - 1; x++) {
      const i = y * w + x;
      const v = 4 * gray[i] - gray[i - w] - gray[i + w] - gray[i - 1] - gray[i + 1];
      sum += v;
      sum2 += v * v;
      n++;
    }
  }
  const mean = sum / n;
  return sum2 / n - mean * mean;
}

function estimateNoise(gray, w, h) {
  let sum = 0, sum2 = 0, n = 0;
  const kernel = [1,2,1, 2,4,2, 1,2,1];
  const kSum = 16;
  for (let y = 1; y < h - 1; y++) {
    for (let x = 1; x < w - 1; x++) {
      let blur = 0, k = 0;
      for (let dy = -1; dy <= 1; dy++) {
        for (let dx = -1; dx <= 1; dx++) {
          blur += gray[(y + dy) * w + (x + dx)] * kernel[k++];
        }
      }
      blur /= kSum;
      const diff = gray[y * w + x] - blur;
      sum += diff;
      sum2 += diff * diff;
      n++;
    }
  }
  const mean = sum / n;
  return Math.sqrt(sum2 / n - mean * mean);
}

function meanAbsDiff(a, b) {
  let s = 0;
  for (let i = 0; i < a.length; i++) s += Math.abs(a[i] - b[i]);
  return s / a.length;
}

function combineSignals(fftMetrics, textureMetrics, tdiff) {
  const avg = arr => arr.reduce((a, b) => a + b, 0) / arr.length;

  const fftPeak = avg(fftMetrics.map(m => m.peak));
  const fftStd  = avg(fftMetrics.map(m => m.std));
  const lapVar  = avg(textureMetrics.map(m => m.laplacianVar));
  const noise   = avg(textureMetrics.map(m => m.noise));

  const tdiffMean = tdiff.length ? avg(tdiff) : 0;
  const tdiffStd = tdiff.length > 1
    ? Math.sqrt(avg(tdiff.map(v => (v - tdiffMean) ** 2)))
    : 0;

  const sigFFTPeak = clip01((fftPeak - 1.6) / 2.4);
  const sigFFTStd  = clip01((fftStd  - 0.30) / 0.55);
  const sigSmooth  = clip01(1 - lapVar / 0.0018);
  const sigNoise   = clip01(1 - noise / 0.035);
  const sigFlicker = clip01(tdiffStd / 0.02);

  const weights = {
    fftPeak: 0.25,
    fftStd:  0.20,
